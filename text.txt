For your Mutualité website, I recommend structuring the About page around the organization’s identity, mission, community initiatives, and impact, while the Donation page should focus on giving opportunities, transparent use of donations, and an interactive donation form.
To satisfy your JavaScript assignment requirements, make both pages share a local JSON data source, ES modules, reusable card components, local storage, and an accessible modal dialog.

Dynamic features for About:
- Load at least 15 program, initiative, or community activity records from JSON.
- Display four or more properties per item, such as title, category, location, and description.
- Add category filters and a search field.
- Include a “Learn More” button opening an accessible modal with additional details.
- Save the visitor’s preferred program category or display setting using localStorage.

Amount · Selected cause · Donor details · Frequency

Show verified funding targets and progress where available

Review the donation details and confirm the next step


Dynamic features for Donations:
- Load 15 or more donation opportunities or giving-related records from JSON.
- Display each cause’s name, category, target amount, and description, with other properties such as location or progress.
- Allow filtering by cause category and selecting a donation amount.
- Update the form summary when the visitor changes their amount or cause.
- Use an accessible modal to review the donation before proceeding.
- Store a draft donation preference locally, but never store payment card details in localStorage.
Important: a front-end donation form does not actually collect money by itself. Real payments require a secure payment provider and a server-side integration. For a coursework demonstration, you can show a clearly labeled demo confirmation without claiming that a payment was completed.

4. Shared JavaScript module: scripts/initiatives.js
This module fetches the JSON data, handles errors using try...catch, and generates cards using template literals. Both pages can import its functions.

const DATA_URL = "./data/initiatives.json";

export async function fetchInitiatives() {
    try {
        const response = await fetch(DATA_URL);

        if (!response.ok) {
            throw new Error(`Request failed: ${response.status}`);
        }

        const initiatives = await response.json();

        if (!Array.isArray(initiatives)) {
            throw new Error("The data format is invalid.");
        }

        return initiatives;
    } catch (error) {
        console.error("Could not load initiatives:", error);
        throw error;
    }
}

export function createInitiativeCard(item, buttonLabel = "Learn More") {
    const card = document.createElement("article");
    card.className = "initiative-card";

    // Build the card safely using DOM methods.
    const title = document.createElement("h3");
    title.textContent = item.title;

    const category = document.createElement("p");
    category.textContent = `Category: ${item.category}`;

    const location = document.createElement("p");
    location.textContent = `Location: ${item.location}`;

    const description = document.createElement("p");
    description.textContent = item.description;

    const target = document.createElement("p");
    target.textContent =
        `Funding target: $${item.targetAmount.toLocaleString("en-US")}`;

    const button = document.createElement("button");
    button.type = "button";
    button.className = "details-button";
    button.textContent = buttonLabel;
    button.dataset.id = item.id;

    card.append(title, category, location, description, target, button);

    return card;
}

export function renderInitiatives(container, items, buttonLabel) {
    container.replaceChildren();

    // Array method: forEach
    items.forEach((item) => {
        container.append(
            createInitiativeCard(item, buttonLabel)
        );
    });
}



Using DOM methods and textContent to insert data helps prevent untrusted JSON text from being interpreted as HTML.

scripts/about.js
This module loads the data, builds the category filter, searches initiatives, and opens a details modal.

import {
    fetchInitiatives,
    renderInitiatives
} from "./initiatives.js";

const list = document.querySelector("#initiative-list");
const search = document.querySelector("#program-search");
const filter = document.querySelector("#category-filter");
const count = document.querySelector("#program-count");
const error = document.querySelector("#program-error");

const dialog = document.querySelector("#details-dialog");
const dialogTitle = document.querySelector("#dialog-title");
const dialogContent = document.querySelector("#dialog-content");
const closeDialog = document.querySelector("#close-dialog");

let initiatives = [];

function displayItems(items) {
    renderInitiatives(list, items, "Learn More");
    count.textContent = `${items.length} initiatives found.`;
}

function applyFilters() {
    const term = search.value.trim().toLowerCase();
    const category = filter.value;

    const filtered = initiatives.filter((item) => {
        const matchesSearch =
            `${item.title} ${item.location} ${item.description}`
                .toLowerCase()
                .includes(term);

        const matchesCategory =
            category === "all" || item.category === category;

        return matchesSearch && matchesCategory;
    });

    displayItems(filtered);
}

async function initializeAboutPage() {
    try {
        initiatives = await fetchInitiatives();

        const categories = [
            ...new Set(initiatives.map((item) => item.category))
        ];

        categories.forEach((category) => {
            const option = document.createElement("option");
            option.value = category;
            option.textContent = category;
            filter.append(option);
        });

        // Restore the visitor's category preference.
        filter.value =
            localStorage.getItem("mutualite-category") || "all";

        if (!categories.includes(filter.value) &&
            filter.value !== "all") {
            filter.value = "all";
        }

        displayItems(initiatives);
        applyFilters();
    } catch {
        error.textContent =
            "Our initiatives could not be loaded. Please try again later.";
    }
}

search.addEventListener("input", applyFilters);

filter.addEventListener("change", () => {
    localStorage.setItem("mutualite-category", filter.value);
    applyFilters();
});

// Event delegation handles buttons on dynamically created cards.
list.addEventListener("click", (event) => {
    const button = event.target.closest(".details-button");

    if (!button) return;

    const item = initiatives.find(
        (initiative) => initiative.id === Number(button.dataset.id)
    );

    if (!item) return;

    dialogTitle.textContent = item.title;
    dialogContent.replaceChildren();

    [
        `Category: ${item.category}`,
        `Location: ${item.location}`,
        item.description,
        `Funding target: $${item.targetAmount.toLocaleString("en-US")}`
    ].forEach((text) => {
        const paragraph = document.createElement("p");
        paragraph.textContent = text;
        dialogContent.append(paragraph);
    });

    dialog.showModal();
});

closeDialog.addEventListener("click", () => dialog.close());

dialog.addEventListener("click", (event) => {
    if (event.target === dialog) dialog.close();
});

initializeAboutPage();

scripts/donation.js

import {
    fetchInitiatives,
    renderInitiatives
} from "./initiatives.js";

const list = document.querySelector("#donation-list");
const search = document.querySelector("#donation-search");
const causeSelect = document.querySelector("#donation-cause");
const form = document.querySelector("#giving-form");
const amountInput = document.querySelector("#donation-amount");
const remember = document.querySelector("#remember-preference");
const summary = document.querySelector("#donation-summary");
const error = document.querySelector("#donation-error");

const dialog = document.querySelector("#donation-dialog");
const reviewDetails = document.querySelector("#review-details");
const editButton = document.querySelector("#edit-donation");
const confirmButton = document.querySelector("#confirm-donation");
const status = document.querySelector("#donation-status");

let initiatives = [];
let pendingDonation = null;

function updateSummary() {
    const cause = initiatives.find(
        (item) => item.id === Number(causeSelect.value)
    );

    const amount = Number(amountInput.value);

    if (!cause || !Number.isFinite(amount) || amount <= 0) {
        summary.textContent = "Choose a cause and enter a valid amount.";
        return;
    }

    summary.textContent =
        `You selected ${cause.title} with a donation of $${amount.toFixed(2)}.`;
}

function showReview(donation) {
    pendingDonation = donation;
    reviewDetails.replaceChildren();

    const lines = [
        `Cause: ${donation.causeTitle}`,
        `Amount: $${donation.amount.toFixed(2)} USD`,
        `Frequency: ${donation.frequency}`,
        `Donor: ${donation.name}`,
        `Email: ${donation.email}`
    ];

    lines.forEach((line) => {
        const paragraph = document.createElement("p");
        paragraph.textContent = line;
        reviewDetails.append(paragraph);
    });

    dialog.showModal();
}

async function initializeDonationPage() {
    try {
        initiatives = await fetchInitiatives();

        // Populate cause selector from fetched data.
        initiatives.forEach((item) => {
            const option = document.createElement("option");
            option.value = item.id;
            option.textContent = item.title;
            causeSelect.append(option);
        });

        renderInitiatives(list, initiatives, "Select This Cause");

        const saved = JSON.parse(
            localStorage.getItem("mutualite-donation-preference") || "null"
        );

        if (saved) {
            if (initiatives.some((item) => item.id === saved.causeId)) {
                causeSelect.value = String(saved.causeId);
            }

            if (Number.isFinite(saved.amount) &&
                saved.amount >= 1 && saved.amount <= 1000000) {
                amountInput.value = String(saved.amount);
                remember.checked = true;
            }
        }

        updateSummary();
    } catch {
        error.textContent =
            "Donation opportunities could not be loaded. Please try again later.";
    }
}

search.addEventListener("input", () => {
    const term = search.value.trim().toLowerCase();

    const filtered = initiatives.filter((item) =>
        `${item.title} ${item.category} ${item.description}`
            .toLowerCase()
            .includes(term)
    );

    renderInitiatives(list, filtered, "Select This Cause");
});

list.addEventListener("click", (event) => {
    const button = event.target.closest(".details-button");

    if (!button) return;

    causeSelect.value = button.dataset.id;
    updateSummary();

    document.querySelector("#donation-form").scrollIntoView({
        behavior: "smooth",
        block: "start"
    });
});

causeSelect.addEventListener("change", updateSummary);
amountInput.addEventListener("input", updateSummary);

form.addEventListener("submit", (event) => {
    event.preventDefault();

    if (!form.reportValidity()) return;

    const amount = Number(amountInput.value);

    if (!Number.isFinite(amount) || amount < 1 || amount > 1000000) {
        amountInput.setCustomValidity("Enter an amount between $1 and $1,000,000.");
        amountInput.reportValidity();
        amountInput.setCustomValidity("");
        return;
    }

    const cause = initiatives.find(
        (item) => item.id === Number(causeSelect.value)
    );

    if (!cause) {
        status.textContent = "Please select a valid cause.";
        return;
    }

    const donation = {
        causeId: cause.id,
        causeTitle: cause.title,
        amount,
        frequency: document.querySelector("#giving-frequency").value,
        name: document.querySelector("#donor-name").value.trim(),
        email: document.querySelector("#donor-email").value.trim()
    };

    if (remember.checked) {
        // Save only non-sensitive preferences, not donor details.
        localStorage.setItem(
            "mutualite-donation-preference",
            JSON.stringify({
                causeId: donation.causeId,
                amount: donation.amount
            })
        );
    } else {
        localStorage.removeItem("mutualite-donation-preference");
    }

    showReview(donation);
});

editButton.addEventListener("click", () => {
    dialog.close();
});

confirmButton.addEventListener("click", () => {
    if (!pendingDonation) return;

    // Coursework demo: this does not process or record a real payment.
    dialog.close();
    status.textContent =
        "Demo confirmed. No payment was made. Thank you for supporting Mutualité.";

    pendingDonation = null;
});

dialog.addEventListener("click", (event) => {
    if (event.target === dialog) dialog.close();
});

initializeDonationPage();
